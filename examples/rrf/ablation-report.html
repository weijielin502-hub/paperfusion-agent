+"""Reciprocal Rank Fusion (RRF).
+
+Each ranked document receives the sum of 1/(k + rank) across the input
+rankings, with ranking positions starting at one (Cormack, Clarke & Buttcher,
+2009). The resulting scores determine descending document order.
+
+Project adaptations: rankings may be truncated (a missing document contributes
+zero for that list); within a list a repeated document ID contributes only at
+its first position, and later duplicates do not shift the retained positions of
+subsequent IDs; ties are broken by ascending document ID.
+"""
+import math
+
+__all__ = ["scores", "fuse"]
+
+
+def _is_real_number(value):
+    return isinstance(value, (int, float)) and not isinstance(value, bool)
+
+
+def _validate_k(k):
+    if not _is_real_number(k):
+        raise ValueError("k must be a finite positive int or float")
+    if isinstance(k, float) and (math.isnan(k) or math.isinf(k)):
+        raise ValueError("k must be a finite positive int or float")
+    if k <= 0:
+        raise ValueError("k must be a finite positive int or float")
+
+
+def _validate_limit(limit):
+    if limit is None:
+        return
+    if not isinstance(limit, int) or isinstance(limit, bool):
+        raise ValueError("limit must be None or a nonnegative int")
+    if limit < 0:
+        raise ValueError("limit must be None or a nonnegative int")
+
+
+def _score_map(rankings, k):
+    """Internal helper: build {doc: rrf_score} for materialized rankings."""
+    result = {}
+    for ranking in rankings:
+        seen = set()
+        for index, doc in enumerate(ranking):
+            if doc in seen:
+                continue
+            seen.add(doc)
+            result[doc] = result.get(doc, 0.0) + 1.0 / (k + (index + 1))
+    return result
+
+

+    """Return {document_id: fused_score} for the given ranked lists."""
+    _validate_k(k)
+    materialized = list(rankings)
+    return _score_map(materialized, k)
